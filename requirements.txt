# TROUBLESHOOTING GUIDE

# Common Installation Issues:

# 1. OpenCV Installation Failed:
# pip install --upgrade pip setuptools wheel
# pip install opencv-python-headless  # Alternative if GUI not needed

# 2. PyTorch CUDA Issues:
# Check CUDA version: nvidia-smi
# Install matching PyTorch version
# Verify GPU availability: python -c "import torch; print(torch.cuda.is_available())"

# 3. ONVIF/Zeep Issues:
# pip install --no-cache-dir onvif-zeep zeep
# Check firewall settings for ONVIF discovery

# 4. yt-dlp Update Issues:
# pip install --upgrade yt-dlp
# Clear cache: yt-dlp --rm-cache-dir

# 5. Permission Issues (Linux):
# sudo pip install -r requirements.txt  # Not recommended
# Use virtual environment instead: python3 -m venv venv && source venv/bin/activate

# 6. Windows-specific Issues:
# Install Microsoft Visual C++ Build Tools
# pip install --upgrade setuptools wheel

# 7. Memory Issues:
# pip install --no-cache-dir -r requirements.txt
# Install packages one by one if needed

# 8. Network/Proxy Issues:
# pip install --trusted-host pypi.org --trusted-host pypi.python.org -r requirements.txt
